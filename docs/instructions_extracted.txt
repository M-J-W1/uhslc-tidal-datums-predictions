# UHSLC Tidal Datums and Predictions
## Purpose: 
Update datums and tide predictions for all Fast Delivery and Research Quality hourly data that is served via ERDDAP.
## Protocol
Learn best practices from legacy software, and make improvements where necessary. The deliverable is a Python script that may be ran in batch mode to cover groups of stations (all, FD, RQ, subsets, or individual stations).
### Output files
Netcdf 
One file per “station” record (e.g., 001, 001a, 001b, 001c would cover Pohnpei for FD and its three RQ versions)
### Reference frames and units
Station zero
GMT
mm integer
### Epochs and data completion
These are the primary epochs: NTDE_1983-2001, NTDE_2002-2020, IPCC-AR6_1995-2014.
Attempt to calculate the datums and tidal harmonics/predictions for each of the primary 19-year epochs. However, this will be impossible for many “station” records, because of their data coverage. Use a 75% of hourly data for a completion criteria. If there is not 75% coverage for a particular epoch, skip it. For all records with at least 6 months of hourly data (>75% complete), make sure to define at least one epoch. That one epoch should be the most recent data, not to exceed 19 years. No station should have more than three epochs. Example: there will be one epoch for 001a (RQ), which will span 1969-05-09 to 1971-02-27.
### DatumsNote that since you do not have access to the switch elevations, it is okay to omit that. Otherwise, learn from the legacy software what datums to calculate and save.
### Tidal harmonics
Harmonics should be calculated for each epoch. Include the tidal harmonics in the output file. Follow best practices learned from the legacy software, especially concerning treatment of the nodal cycle and trends.
### Tide predictions
Tide predictions should be calculated for each set of harmonics. For FD stations: Output hourly predictions spanning the epoch start to 2035-12-31 23 hour. Also calculate minute predictions from 2025-01-01 00 hour to 2030-12-31 23:59 hour (FD stations only). To reduce file sizes, only the times and heights of high/low daily tides from the minute predictions should be saved (review the legacy procedure). For the RQ stations: only calculate and save the hourly tide predictions spanning the available epochs (e.g., the prediction for 001a will cover the hours spanning 1969-05-09 to 1971-02-27).   
### Test stations Run tests on generate plots for the following stations, FD and all RQ versions: 001, 002, 003, and 007 (datums, hourly predictions/residuals, and minute predictions if application).

%% Tidal Datums
%  https://tidesandcurrents.noaa.gov/datum_options.html
for station_ct = 1:size(uhidTxt,1)
  if ~isnan(STATION(station_ct).latitude{1})
    
    %% Generated by SEA to more closely match NOAA's method using a tidal day (MJW: 7/14/2025)
    disp(strcat('Calculating Tidal Datums (Tidal Day Window): ',STATION(station_ct).station_UHSLC));
    sl_NTDE = STATION(station_ct).sl_NTDE(:,1);
    t_NTDE = STATION(station_ct).t_NTDE(:,1);
    % Remove NaNs
    valid_idx = ~isnan(sl_NTDE) & ~isnan(t_NTDE);
    sl_NTDE = sl_NTDE(valid_idx);
    t_NTDE = t_NTDE(valid_idx);
    if numel(sl_NTDE) > t_min
    %% --- MHW (Mean High Water) and MLW (Mean Low Water) using all local maxima/minima ---
    [~, locs_high] = findpeaks(sl_NTDE, 'MinPeakDistance', 6); % Find all local maxima (high waters, at least 6 hours between peaks)
    [~, locs_low]  = findpeaks(-sl_NTDE, 'MinPeakDistance', 6); % Find all local minima (low waters, at least 6 hours between troughs)
    HW = sl_NTDE(locs_high);
    LW = sl_NTDE(locs_low);
    MHW_NTDE = mean(HW);
    MLW_NTDE = mean(LW);
    STATION(station_ct).MHW_NTDE = MHW_NTDE;
    STATION(station_ct).MLW_NTDE = MLW_NTDE;  
         
    %% --- TIDETYPE: Diurnal or Mixed/Semidiurnal (using high tide intervals) ---
    t_highs = t_NTDE(locs_high); % Use high waters calculated above
    t_delta = diff(t_highs); % In days (assuming t_NTDE is in days)
    if nanmean(t_delta) > 0.75
        diurnal_flag = 1;
        STATION(station_ct).TideType = 'Diurnal';
    else
        diurnal_flag = 0;
        STATION(station_ct).TideType = 'Semidiurnal/Mixed';
    end
    %% --- MHHW, MLLW, MSL using running tidal day window (24h50m = 1.0347 days) ---
    tidal_day_days = (24 + 50/60) / 24;
    t_start = t_NTDE(1);
    t_end = t_NTDE(end);
    %window_starts = t_start : 1 : (t_end - tidal_day_days);
    window_starts = t_start : tidal_day_days : (t_end - tidal_day_days); % Avoids overlapping bins
    n_windows = length(window_starts);
    max_list = nan(n_windows,1);
    min_list = nan(n_windows,1);
    mean_list = nan(n_windows,1);
    for i = 1:n_windows
        t0 = window_starts(i);
        t1 = t0 + tidal_day_days;
        idx = t_NTDE >= t0 & t_NTDE < t1;
        window_data = sl_NTDE(idx);
        if sum(~isnan(window_data)) >= 20 % Checks if there are at least 20 valid (non-NaN) data points in the window.
            max_list(i) = max(window_data);
            min_list(i) = min(window_data);
            mean_list(i) = mean(window_data);
        end
    end
    % Remove NaNs
    max_list = max_list(~isnan(max_list));
    min_list = min_list(~isnan(min_list));
    mean_list = mean_list(~isnan(mean_list));
    % Calculate datums
    MHHW_NTDE = mean(max_list);
    MLLW_NTDE = mean(min_list);
    %MSL_NTDE  = mean(mean_list); % Calculated below as simple average of the NTDE time series
    STATION(station_ct).MHHW_NTDE = MHHW_NTDE;
    STATION(station_ct).MLLW_NTDE = MLLW_NTDE;
    %STATION(station_ct).MSL_NTDE  = MSL_NTDE; % Calculated below as simple average of the NTDE time series
    % -------------------------------------------
    % Save timeseries referenced to MLLW datums:
    sl_NTDE_MLLW = sl_NTDE - MLLW_NTDE;   
    % Display datums
    disp(['MHHW (tidal day window): ', num2str(MHHW_NTDE), ' m']);
    disp(['MLLW (tidal day window): ', num2str(MLLW_NTDE), ' m']);
    %disp(['MSL  (tidal day window): ', num2str(MSL_NTDE),  ' m']);
    disp(['MHW  (all highs): ', num2str(MHW_NTDE),  ' m']);
    disp(['MLW  (all lows): ', num2str(MLW_NTDE),  ' m']);
    
    % -------------------------------------------
    % Higher High Water (HHW) and Lower Low Water (LLW) timeseries using tidal day windows
    % max_list and min_list are already calculated above as the max/min in each tidal day window
    STATION(station_ct).HHW_NTDE_MLLW = max_list - MLLW_NTDE; % Timeseries of HHW referenced to MLLW
    STATION(station_ct).LLW_NTDE_MLLW = min_list - MLLW_NTDE; % Timeseries of LLW referenced to MLLW
    clear max_list min_list MLLW_NTDE
      %% -------------------------------------------
      %% Mean Diurnal High/Low Water Inequality (Revised 11/14/2022)
      % One-half the average difference between the two high/low waters of each tidal day observed over the National Tidal Datum Epoch. 
      % It is obtained by subtracting the mean of all the high waters from the mean of the higher high waters.
      STATION(station_ct).DHQ_NTDE = STATION(station_ct).MHHW_NTDE - STATION(station_ct).MHW_NTDE; 
      
      % It is obtained by subtracting the mean of the lower low waters from the mean of all the low waters.
      STATION(station_ct).DLQ_NTDE =  STATION(station_ct).MLW_NTDE - STATION(station_ct).MLLW_NTDE;
      
      %% -------------------------------------------
      %% Diurnal Tide Level (DTL)
      DTL_NTDE = (STATION(station_ct).MHHW_NTDE + STATION(station_ct).MLLW_NTDE)/2;
      STATION(station_ct).DTL_NTDE = DTL_NTDE; 
      clear('DTL_NTDE');
      
      %% -------------------------------------------
      %% Mean Tide Level (MTL)
      MTL_NTDE = (STATION(station_ct).MHW_NTDE + STATION(station_ct).MLW_NTDE)/2;
      STATION(station_ct).MTL_NTDE = MTL_NTDE; 
      clear('MTL_NTDE');
      
      %% -------------------------------------------
      %% Mean Sea Level (MSL)
      MSL_NTDE = nanmean(sl_NTDE,1);
      STATION(station_ct).MSL_NTDE = MSL_NTDE; 
      clear('MSL_NTDE');
      
      %% -------------------------------------------
      %% Great Diurnal Range (GT)
      GT_NTDE = STATION(station_ct).MHHW_NTDE - STATION(station_ct).MLLW_NTDE;
      STATION(station_ct).GT_NTDE = GT_NTDE;
      clear('GT_NTDE');
      
      %% -------------------------------------------
      %% Mean Range of Tide (MN)
      MN_NTDE = STATION(station_ct).MHW_NTDE - STATION(station_ct).MLW_NTDE;
      STATION(station_ct).MN_NTDE = MN_NTDE;
      clear('MN_NTDE');
      
      %% -------------------------------------------
      %% Greenwich High Water Interval (HWI)
      % The average interval (in hours) between the moon's transit over the Greenwich meridian and the following high water at a location.
      %  incomplete...
      STATION(station_ct).HWI_NTDE = NaN;
      
      %% -------------------------------------------
      %% Greenwich Low Water Interval (LWI)
      % The average interval (in hours) between the moon's transit over the Greenwich meridian and the following low water at a location.
      %  incomplete...
      STATION(station_ct).LWI_NTDE = NaN;
      %% -------------------------------------------
      %% Extremes (90%, 95%, 99%)    
      thresholds = [100-90, 100-95, 100-99];
      thresholds_txt = {'90','95','99'};
      for thres_ct = 1:size(thresholds,2)
        %% Hourly
        tmp_extrmLow = prctile(sl_NTDE_MLLW(:,1),thresholds(thres_ct));
        tmp_extrmHigh = prctile(sl_NTDE_MLLW(:,1),100-thresholds(thres_ct));
        name_extrmLow = ['SLntde_extrmLow_',thresholds_txt{thres_ct}]; 
        name_extrmHigh = ['SLntde_extrmHigh_',thresholds_txt{thres_ct}];
        STATION(station_ct).(name_extrmLow) = tmp_extrmLow;
        STATION(station_ct).(name_extrmHigh) = tmp_extrmHigh;
        clear tmp_extrmLow tmp_extrmHigh name_extrmLow name_extrmHigh
        
        %% Daily (Lower Low Water and Higher High Water)
        tmp_extrmLow = prctile(STATION(station_ct).LLW_NTDE_MLLW(:,1),thresholds(thres_ct));
        tmp_extrmHigh = prctile(STATION(station_ct).HHW_NTDE_MLLW(:,1),100-thresholds(thres_ct));
        name_extrmLow = ['LLW_NTDE_MLLW_',thresholds_txt{thres_ct}]; 
        name_extrmHigh = ['HHW_NTDE_MLLW_',thresholds_txt{thres_ct}];
        STATION(station_ct).(name_extrmLow) = tmp_extrmLow;
        STATION(station_ct).(name_extrmHigh) = tmp_extrmHigh;
        clear tmp_extrmLow tmp_extrmHigh name_extrmLow name_extrmHigh
      end
      clear thresholds thresholds_txt
      %% -------------------------------------------
      clear('diurnal_flag','sl_NTDE','HighTides','LowTides');
    else
      disp(strcat('No Hourly Tide Gauge Data during NTDE: ',STATION(station_ct).station_UHSLC));
      STATION(station_ct).TideType = '';
      STATION(station_ct).MHHW_NTDE = NaN;
      STATION(station_ct).MHW_NTDE = NaN;
      STATION(station_ct).DTL_NTDE = NaN;
      STATION(station_ct).MTL_NTDE = NaN;
      STATION(station_ct).MSL_NTDE = NaN;
      STATION(station_ct).MLW_NTDE = NaN;
      STATION(station_ct).MLLW_NTDE = NaN;     
      STATION(station_ct).GT_NTDE = NaN;
      STATION(station_ct).MN_NTDE = NaN;
      STATION(station_ct).DHQ_NTDE = NaN;
      STATION(station_ct).DLQ_NTDE = NaN;
      STATION(station_ct).HWI_NTDE = NaN;
      STATION(station_ct).LWI_NTDE = NaN;
      
      %STATION(station_ct).sl_NTDE_MLLW(:,1) = NaN;
      
      STATION(station_ct).SLntde_extrmLow_90 = NaN;
      STATION(station_ct).SLntde_extrmLow_95 = NaN;
      STATION(station_ct).SLntde_extrmLow_99 = NaN;
      
      STATION(station_ct).SLntde_extrmHigh_90 = NaN;
      STATION(station_ct).SLntde_extrmHigh_95 = NaN;
      STATION(station_ct).SLntde_extrmHigh_99 = NaN;
      
      STATION(station_ct).LLW_NTDE_MLLW_90 = NaN;
      STATION(station_ct).LLW_NTDE_MLLW_95 = NaN;
      STATION(station_ct).LLW_NTDE_MLLW_99 = NaN;
      
      STATION(station_ct).HHW_NTDE_MLLW_90 = NaN;
      STATION(station_ct).HHW_NTDE_MLLW_95 = NaN;
      STATION(station_ct).HHW_NTDE_MLLW_99 = NaN;
    end    
    
  else
    disp(strcat('No Hourly Tide Gauge Data: ',STATION(station_ct).station_UHSLC));
    STATION(station_ct).TideType = '';
    STATION(station_ct).MHHW_NTDE = NaN;
    STATION(station_ct).MHW_NTDE = NaN;
    STATION(station_ct).DTL_NTDE = NaN;
    STATION(station_ct).MTL_NTDE = NaN;
    STATION(station_ct).MSL_NTDE = NaN;
    STATION(station_ct).MLW_NTDE = NaN;
    STATION(station_ct).MLLW_NTDE = NaN;     
    STATION(station_ct).GT_NTDE = NaN;
    STATION(station_ct).MN_NTDE = NaN;
    STATION(station_ct).DHQ_NTDE = NaN;
    STATION(station_ct).DLQ_NTDE = NaN;
    STATION(station_ct).HWI_NTDE = NaN;
    STATION(station_ct).LWI_NTDE = NaN;
    %STATION(station_ct).sl_NTDE_MLLW(:,1) = NaN;
    
    STATION(station_ct).SLntde_extrmLow_90 = NaN;
    STATION(station_ct).SLntde_extrmLow_95 = NaN;
    STATION(station_ct).SLntde_extrmLow_99 = NaN;

% Parameters for UT_solv
%anal = 'annual'; % Analysis option (year-by-year)
anal = 'epoch'; % Analysis option (all years together)
%opt = 'stats'; % Full analysis power
opt = 'nostats'; % Less analysis power (but still good)

MinSNR = 2; % Same as utide default

for station_ct = 1:size(uhidTxt,1)
  if ~isnan(STATION(station_ct).latitude{1})  % Tide gauge exists
    t = STATION(station_ct).t_NTDE(:,1);
    y = STATION(station_ct).sl_NTDE(:,1);
    
    % Done above

    t_datenum = t + datenum(1800,1,1); % Convert to "datenum" format (days since Jan-1-0000)

    STATION(station_ct).nHr_NTDE = size(y_noNaN,1); % Number of hours available during NTDE

    STATION(station_ct).results = NaN;
    STATION(station_ct).yres = NaN;
  end
  clear('t','t_datenum','y','y_noNaN','lat','ytide','results','yres');

%% Calculate Extremes (Highest & Lowest Predicted Astronomical Tides)
for station_ct = 1:size(uhidTxt,1)
  MLLW_NTDE = STATION(station_ct).MLLW_NTDE;
  t = STATION(station_ct).t_NTDE(:,1); 
  results = STATION(station_ct).results;
  
  t_datenum = t + datenum(1800,1,1); % Convert to "datenum" format (days since Jan-1-0000) 
  if isfield(results,'coef') % Coefficients exist
    disp(strcat('Calculating Extremes: ',STATION(station_ct).station_UHSLC));
    
    % Ignore trend
    results_NoTrend = results;
    results_NoTrend.coef.slope = 0;
    
    % ----------------------------
    % Tide prediction (NTDE epoch)
    tic
    [ sl_fit_NTDE, ~ ] = ut_reconstr(t_datenum, results_NoTrend.coef); 
    toc
    sl_fit_NTDE_MLLW = (sl_fit_NTDE(:,1) - MLLW_NTDE);
    
    % Highest Astronomical Tide (HAT) & Time
    [HAT_NTDE i_HAT_NTDE] = max(sl_fit_NTDE);
    STATION(station_ct).HAT_NTDE = HAT_NTDE;
    t_HAT_NTDE = STATION(station_ct).t_NTDE(i_HAT_NTDE,1);
    STATION(station_ct).t_HAT_NTDE = datestr(t_HAT_NTDE+datenum(1800,1,1));
    clear('HAT_NTDE','t_HAT_NTDE');
    
    % Lowest Astronomical Tide (LAT) & Time
    [LAT_NTDE i_LAT_NTDE] = min(sl_fit_NTDE);
    STATION(station_ct).LAT_NTDE = LAT_NTDE;
    t_LAT_NTDE = STATION(station_ct).t_NTDE(i_LAT_NTDE,1);
    STATION(station_ct).t_LAT_NTDE = datestr(t_LAT_NTDE+datenum(1800,1,1));
    clear('LAT_NTDE','t_LAT_NTDE');
    % ----------------------------
    % Find High and Low tides (NTDE epoch)
    delta = 10^-6;
    [HighTides_NTDE_MLLW, LowTides_NTDE_MLLW] = peakdet(((sl_fit_NTDE(:,1) - MLLW_NTDE)),delta);
    % ----------------------------
    % Find Extreme High and Low tides (NTDE epoch)
    
    %% Legacy extremes (High and low tide predictions)
    threshold = 5; % Top and bottom %
    sl_fit_NTDE_MLLW_extrmLow = prctile(LowTides_NTDE_MLLW(:,2),threshold);
    sl_fit_NTDE_MLLW_extrmHigh = prctile(HighTides_NTDE_MLLW(:,2),100-threshold);
    
    STATION(station_ct).sl_fit_NTDE_MLLW = sl_fit_NTDE_MLLW;
    STATION(station_ct).sl_fit_NTDE_MLLW_extrmLow = sl_fit_NTDE_MLLW_extrmLow;
    STATION(station_ct).sl_fit_NTDE_MLLW_extrmHigh = sl_fit_NTDE_MLLW_extrmHigh;
    
    % New: 90%, 95%, 99%
    thresholds = [100-90, 100-95, 100-99];
    thresholds_txt = {'90','95','99'};
    for thres_ct = 1:size(thresholds,2)
      tmp_extrmLow = prctile(LowTides_NTDE_MLLW(:,2),thresholds(thres_ct));
      tmp_extrmHigh = prctile(HighTides_NTDE_MLLW(:,2),100-thresholds(thres_ct));
      
      name_extrmLow = ['Tide_extrmLow_',thresholds_txt{thres_ct}]; 
      name_extrmHigh = ['Tide_extrmHigh_',thresholds_txt{thres_ct}];
      
      STATION(station_ct).(name_extrmLow) = tmp_extrmLow;
      STATION(station_ct).(name_extrmHigh) = tmp_extrmHigh;
      
      clear tmp_extrmLow tmp_extrmHigh name_extrmLow name_extrmHigh
    end
    clear thresholds thresholds_txt
      
  else
    disp(strcat('No Tide Coefficients: ',STATION(station_ct).station_UHSLC));
        
    STATION(station_ct).sl_fit_NTDE_MLLW = NaN;
    STATION(station_ct).sl_fit_NTDE_MLLW_extrmLow = NaN;
    STATION(station_ct).sl_fit_NTDE_MLLW_extrmHigh = NaN;
    
    STATION(station_ct).HAT_NTDE = NaN;
    STATION(station_ct).t_HAT_NTDE = NaN;
    STATION(station_ct).LAT_NTDE = NaN;
    STATION(station_ct).t_LAT_NTDE = NaN;
    
    % New: 90%, 95%, 99%
    thresholds = [100-90, 100-95, 100-99];
    thresholds_txt = {'90','95','99'};
    for thres_ct = 1:size(thresholds,2)  
      name_extrmLow = ['Tide_extrmLow_',thresholds_txt{thres_ct}]; 
      name_extrmHigh = ['Tide_extrmHigh_',thresholds_txt{thres_ct}];
      
      STATION(station_ct).(name_extrmLow) = NaN;
      STATION(station_ct).(name_extrmHigh) = NaN;
      
      clear name_extrmLow name_extrmHigh
    end
    clear thresholds thresholds_txt
    
  end
  clear('t','results','sl_fit_NTDE','sl_fit_NTDE_MLLW','HighTides_NTDE_MLLW','LowTides_NTDE_MLLW',...
    'sl_fit_NTDE_MLLW_extrmLow','sl_fit_NTDE_MLLW_extrmHigh');
  
end

      [ sl_fit, ~ ] = ut_reconstr(t_fit_HH, results_NoTrend.coef); 
      STATION(station_ct).time_PRE_FUTURE(:,1) = t_fit_HH(1,:)' - datenum(1800,1,1); % Index(Hours since 1800), Values(Days or Hours since 1800)  
      STATION(station_ct).tide_PRE_FUTURE(:,1) = sl_fit(1,:)'; % Index(Hours since 1800), Values(Harmonic Tidal Prediction)   
    end
    
    %% Save Tide Predictions as CSV (added 10/28/2024)
    %csv_filename = strcat(dir_path,uhidTxt{station_ct},'_TidePrediction_hourly_mm_StationZero_',num2str(preStartYr),'_',num2str(preEndYr),'.csv');
    idx_avail = find(~isnan(STATION(station_ct).t_NTDE(:,1)));
    pred_epoch_start_txt = datestr(STATION(station_ct).t_NTDE(idx_avail(1),1)+datenum(1800,1,1),'dd-mmm-yyyy');
    pred_epoch_end_txt = datestr(STATION(station_ct).t_NTDE(idx_avail(end),1)+datenum(1800,1,1),'dd-mmm-yyyy');
    csv_filename = strcat(dir_path,uhidTxt{station_ct},'_TidePrediction_hourly_mm_StationZero_',num2str(preStartYr),'_',num2str(preEndYr),'_BasedOn_',pred_epoch_start_txt,'_',pred_epoch_end_txt,'.csv');
    clear idx_avail pred_epoch_start_txt pred_epoch_end_txt
    
    % Format
    reference_datum = 0 - 0; % No conversion (Zero to Zero) Reference datum
    time_zone_opt = 0; % GMT
    units_opt = convert_ssh; % Convert from meters back to mm
    time_data = datestr(STATION(station_ct).time_PRE_FUTURE + datenum(1800,1,1) + datenum(0,0,0,time_zone_opt,0,0),'dd-mmm-yyyy HH');
    tide_data = (STATION(station_ct).tide_PRE_FUTURE - reference_datum) * units_opt; 
    if units_opt == 1000 % mm
      % Round to nearest integer
      tide_data = round(tide_data);
    end
    
    % Combine time and tide predictions into a table
    TidePredictionTable = table(time_data, tide_data, 'VariableNames', {'Time_GMT', 'TidePrediction_mm'});
    % Write to CSV
    writetable(TidePredictionTable, csv_filename);
    clear id_txt csv_filename time_data tide_data TidePredictionTable
 
  end

%% Tidal Prediction (Multi-years)
disp('Calculating Tidal Prediction');
% preStartYr = 2016;
% preEndYr = 2020;
% preStartYr = 2018;
% preEndYr = 2023;
% preStartYr = 2020;
% preEndYr = 2026;
preStartYr = 2023;
preEndYr = 2029;
for station_ct = 1:size(uhidTxt,1)
%for station_ct = tide_partial_start:size(uhidTxt,1)  % Station subgroup only
%for station_ct = tide_partial_set % Station subgroup only
  disp(STATION(station_ct).station_UHSLC);
  
  results = STATION(station_ct).results;
  
  if isfield(results,'coef') % Coefficients exist
  
    % Ignore trend
    results_NoTrend = results;
    results_NoTrend.coef.slope = 0;
    t_fit_HHMM = [datenum(preStartYr,1,1,0,0,0):(1/(24*60)):datenum(preEndYr+1,1,1,0,0,0)-(1/(24*60))]; % Minutes
    %t_fit_HH = [datenum(preStartYr,1,1,0,0,0):(1/24):datenum(preEndYr+1,1,1,0,0,0)-(1/24)]; % Hours
    disp(strcat('Calling "ut_reconstr" (HH): ',num2str(preStartYr),'-',num2str(preEndYr)));
    [ sl_fit, ~ ] = ut_reconstr(t_fit_HHMM, results_NoTrend.coef); 
%     STATION(station_ct).time_PRE(:,1) = t_fit_HHMM(1,:)' - datenum(1800,1,1); % Index(Hours since 1800), Values(Days or Hours since 1800)  
%     STATION(station_ct).tide_PRE(:,1) = sl_fit(1,:)'; % Index(Hours since 1800), Values(Harmonic Tidal Prediction)   
    clear time_PRE tide_PRE
    time_PRE(:,1) = t_fit_HHMM(1,:)' - datenum(1800,1,1); % Index(Hours since 1800), Values(Days or Hours since 1800)  
    tide_PRE(:,1) = sl_fit(1,:)'; % Index(Hours since 1800), Values(Harmonic Tidal Prediction)  
    
    % ----------------------------
    % Find High and Low tides
    delta = 10^-6;
    MLLW_NTDE = STATION(station_ct).MLLW_NTDE;
    [HighTides, LowTides] = peakdet(((sl_fit- MLLW_NTDE)),delta); % Relative to MLLW
    
    time_diff = 3*60; % Check for "false" tide at start (3hr)
    if HighTides(1,1) < LowTides(1,1) % High Tide first
      if abs(HighTides(1,1)-LowTides(1,1))<time_diff
        HighTides_fix = HighTides(2:end,:);
        clear('HighTides');
        HighTides = HighTides_fix;
        clear('HighTides_fix');
      end
    elseif LowTides(1,1) < HighTides(1,1) % Low Tide first
      if abs(LowTides(1,1)-HighTides(1,1))<time_diff
        LowTides_fix = LowTides(2:end,:);
        clear('LowTides');
        LowTides = LowTides_fix;
        clear('LowTides_fix');
      end
    end
    t_HighTides = t_fit_HHMM(HighTides(:,1));
    t_LowTides = t_fit_HHMM(LowTides(:,1));
    % Local Standard Time
    time_zone = STATION(station_ct).time_zone;
    if isempty(time_zone) % Set as GMT
      time_zone = 0;
    end
    
  %     % Greenwich Mean Time
  %     time_zone = 0;
    
    if time_zone>0
      time_zone_txt = ['Local Standard Time (LST = GMT +',num2str(time_zone),' hr)'];
      time_zone_txt_abb = ['(LST = GMT +',num2str(time_zone),' hr)'];
    elseif time_zone<0
      time_zone_txt = ['Local Standard Time (LST = GMT ',num2str(time_zone),' hr)'];
      time_zone_txt_abb = ['(LST = GMT ',num2str(time_zone),' hr)'];
    else
      time_zone_txt = ['Greenwich Mean Time'];
      time_zone_txt_abb = ['(GMT)'];
    end
    % Convert Time
    t_HighTides_FORMAT = datestr(t_HighTides + datenum(0,0,0,time_zone,0,0),'dd-mmm-yyyy HH:MM');
    t_LowTides_FORMAT = datestr(t_LowTides + datenum(0,0,0,time_zone,0,0),'dd-mmm-yyyy HH:MM');
    t_fit_HHMM_FORMAT = t_fit_HHMM + datenum(0,0,0,time_zone,0,0);
    t_fit_HHMM_FORMAT_datestr = datestr(t_fit_HHMM + datenum(0,0,0,time_zone,0,0),'dd');
    
    n_TIDE = size(t_fit_HHMM_FORMAT,2); % Number of Minutes
    time_month_TIDE = (([1:n_TIDE])/1 + 0);     % construct time array
    xlim_month_TIDE = [min(time_month_TIDE(1,1)), max(time_month_TIDE(1,end))];  % plotting range
    zero_line_TIDE = zeros(n_TIDE,1);
    % ----------------------------
    % Make Tide Table
    TideTable = cell(size(t_HighTides_FORMAT,1)+size(t_LowTides_FORMAT,1),2); % (Times,Tides)
    TideTable(1:end,1:end) =cellstr('NaN');
    if (t_HighTides(1)-t_LowTides(1)) <0 % High Tide first
      TideTable(1:2:end-1,1)= cellstr(t_HighTides_FORMAT(1:size(TideTable(1:2:end-1,1)),:)); 
      %TideTable(1:2:end-1,2) = cellstr(num2str(roundn(HighTides(1:size(TideTable(1:2:end-1,1)),2),0)));
      TideTable(1:2:end-1,2) = cellstr(num2str(roundn(HighTides(1:size(TideTable(1:2:end-1,1)),2),-2))); % m
      TideTable(2:2:end,1) = cellstr(t_LowTides_FORMAT(1:size(TideTable(1:2:end-1,1)),:));
      %TideTable(2:2:end,2) = cellstr(num2str(roundn(LowTides(1:size(TideTable(1:2:end-1,1)),2),0)));
      TideTable(2:2:end,2) = cellstr(num2str(roundn(LowTides(1:size(TideTable(1:2:end-1,1)),2),-2))); % m
      
    else  % Low Tide first
      TideTable(1:2:end-1,1) = cellstr(t_LowTides_FORMAT(1:size(TideTable(1:2:end-1,1)),:));
      %TideTable(1:2:end-1,2) = cellstr(num2str(roundn(LowTides(1:size(TideTable(1:2:end-1,1)),2),0)));
      TideTable(1:2:end-1,2) = cellstr(num2str(roundn(LowTides(1:size(TideTable(1:2:end-1,1)),2),-2))); % m
      TideTable(2:2:end,1) = cellstr(t_HighTides_FORMAT(1:size(TideTable(2:2:end,1)),:));
      %TideTable(2:2:end,2) = cellstr(num2str(roundn(HighTides(1:size(TideTable(2:2:end,1)),2),0)));
      TideTable(2:2:end,2) = cellstr(num2str(roundn(HighTides(1:size(TideTable(2:2:end,1)),2),-2))); % m
    end
    % ----------------------------
    % Flag extreme tides & Water Levels
    ExtrmLow_NTDE_MLLW = STATION(station_ct).sl_fit_NTDE_MLLW_extrmLow;
    ExtrmHigh_NTDE_MLLW = STATION(station_ct).sl_fit_NTDE_MLLW_extrmHigh;
    
    for ct = 1:size(TideTable,1)
      % Tides
      if (str2num(cell2mat(TideTable(ct,2))) <= ExtrmLow_NTDE_MLLW)|(str2num(cell2mat(TideTable(ct,2))) >= ExtrmHigh_NTDE_MLLW) 
        TideTable(ct,2) = strcat(TideTable(ct,2),'*');
      end
    end
    % ----------------------------
end
